cmake_minimum_required(VERSION 3.22)
project(mindinguflac_lyrics_alignment LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    # Inference must stay optimized even in a debug APK.
    add_compile_options($<$<CONFIG:Debug>:-O3>)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
if(NOT WHISPER_SOURCE_DIR)
    get_filename_component(WHISPER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../build/lyrics-whisper/source" ABSOLUTE)
endif()
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper EXCLUDE_FROM_ALL)
add_library(lyrics_whisper SHARED alignment.cpp)
target_link_libraries(lyrics_whisper PRIVATE whisper)
if(ANDROID)
    target_link_options(lyrics_whisper PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(MSVC)
    target_compile_options(lyrics_whisper PRIVATE /utf-8)
endif()
